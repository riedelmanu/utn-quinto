MODEL:
SETS:
ME/M1, M2, M3, F/;
M(ME)/M1, M2, M3/;
TH/HN, HE/;
ORIGENES(M,TH): OFERTAS;
DESTINOS(ME): DEMANDAS;
RUTAS (ORIGENES, DESTINOS): COSTOS, X;
ENDSETS
DATA:
OFERTAS = 1000 400 1200 400 1200 400;
DEMANDAS = 1000 1500 1800 300;
COSTOS = 4 5.5 7 0
6 7.5 9 0
99 4 5.5 0
99 6 7.5 0
99 99 4 0
99 99 6 0;
ENDDATA
MIN = @SUM(RUTAS(i, t, j): COSTOS(i, t, j) * X(i, t, j));
!Capacidades;
@FOR(ORIGENES(i,t):
@SUM(DESTINOS(j): X(i,t,j)) = OFERTAS(i,t));
!Demandas;
@FOR(DESTINOS(j):
@SUM(ORIGENES(i,t): X(i,t,j)) = DEMANDAS(j));
END
